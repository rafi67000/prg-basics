
## Skrypt który użyłem

`cat 1-input | cut -d " " -f 1 | xargs -I {} sh -c 'echo "{}: \n name = {} \n value = $(python3 -c "$(cat 1-input); print({})")" \\n type = $(python3 -c "$(cat 1-input); print(type({}).__name__)")'`

```python
#company: 
 print("company") 
 print(company) 
 print(type(company).__name__)
#phone: 
 print("phone") 
 print(phone) 
 print(type(phone).__name__)
#employees: 
 print("employees") 
 print(employees) 
 print(type(employees).__name__)
#remote_work: 
 print("remote_work") 
 print(remote_work) 
 print(type(remote_work).__name__)
#big_company: 
 print("big_company") 
 print(big_company) 
 print(type(big_company).__name__)
#income: 
 print("income") 
 print(income) 
 print(type(income).__name__)
#income_per_person: 
 print("income_per_person") 
 print(income_per_person) 
 print(type(income_per_person).__name__)
```
